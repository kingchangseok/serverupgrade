int main(argc, argv)
{
    // ====== 1단계: 파라미터 파싱 (398-430) ======
    for (i=1; i<argc; i++) {
        if (!strncmp(argv[i], "PORT=", 5))
            Port = atoi(&argv[i][5]);        // 기본값: 29895
        
        if (!strncmp(argv[i], "LOG=", 4)) {
            if (!strncmp(&argv[i][4], "TRUE", 4))

        }

        if (!strncmp(argv[i], "BUFFER=", 7))
            dfMaxFileSize = atoi(&argv[i][7]); // 버퍼 크기
    }
    
    // ====== 2단계: 시그널 핸들러 등록 (432) ======
    DefineParentSignal();
    
    // ====== 3단계: 서버 소켓 생성 (434-445) ======
    PSock = ServerSocketInit(Port);
    if (PSock < 0) {
        fprintf(stderr, "Socket Create Error PORT[%d]\n", Port);
        exit(1);
    }
    
    // ====== 4단계: 메인 루프 - 연결 대기 (453-478) ======
    while (1) {
        CSock = ServerAcceptClient(PSock);  // 블로킹 대기
        
        if (CSock < 0)
            continue;
        
        // ===== 5단계: Fork (465-478) =====
        if ((forkpid = fork()) < 0) {
            continue;  // ⚠️ 에러 처리 없음!
        }
        else if (!forkpid) {
            // 자식 프로세스
            close(PSock);                    // 부모 소켓 닫기
            DefineChildSignal();             // 자식 시그널 핸들러
            HandleMainProc(CSock);           // 요청 처리
            close(CSock);
            ExitChildProc(0);
        }
        else {
            // 부모 프로세스
            close(CSock);                    // 자식 소켓 닫기
            // 계속 accept() 대기
        }
    }
}

### 1.3 시그널 핸들러

#### 부모 프로세스 (249-274)
```c
void DefineParentSignal()
{
    // 모든 시그널 무시
    for (sig_num = 1; sig_num < 32; sig_num++)
        signal(sig_num, SIG_IGN);
    
    // 선택적 핸들러 등록
    signal(SIGCHLD, SIG_IGN);      // ⚠️ 좀비 프로세스!
    signal(SIGBUS,  ExitParentProc);
    signal(SIGSEGV, ExitParentProc);
    signal(SIGTERM, ExitParentProc);
    signal(SIGINT,  ExitParentProc);
}
```

#### 자식 프로세스 (290-306)
```c
void DefineChildSignal()
{
    for (sig_num = 0; sig_num < 32; sig_num++)
        signal(sig_num, SIG_IGN);
    
    signal(SIGBUS,  ExitChildProc);
    signal(SIGSEGV, ExitChildProc);
    signal(SIGTERM, ExitChildProc);
    signal(SIGINT,  ExitChildProc);
    signal(SIGPIPE, ExitChildProc);  // Broken pipe 처리
}
```

**주의**: SIGCHLD를 `SIG_IGN`으로 설정하면 **좀비 프로세스가 누적됨**!

int ProcSystemCmd(Sock, RdBuf, RdLng)
{
    // 1. AES 키 하드코딩 (⚠️ 보안 문제!)
    u8 szKey[16] = { 0x06, 0xa9, 0x21, 0x40, ... };
    u8 szIV[16]  = { 0x3d, 0xaf, 0xba, 0x42, ... };
    
    // 2. 암호화된 명령어 추출
    memcpy(tmpbuf, &RdBuf[hdlng], pulng);
    
    // 3. AES-128 CBC 복호화
    aes_128_cbc_decrypt(szKey, szIV, (u8 *)decbuf, ncount);
    
    // 4. system() 직접 실행 (⚠️ RCE 위험!)
    retval = system(decbuf) / 256;
    
    // 5. 결과 반환
    if (!retval)
        WorkResultSendData(Sock, tmpbuf, dfCmd_DAOK);
    else
        WorkResultSendData(Sock, tmpbuf, dfCmd_EROR);
}

int GetFileInf(char *FileName)
{
    // stat() 호출
    if (stat(FileName, &f_st) < 0)
        return (1);
    
    // 1. 파일 크기
    gFileSize = f_st.st_size;
    
    // 2. 수정 시간
    loc = localtime(&f_st.st_mtime);
    sprintf(gFileDate, "%d%002d%002d%002d%002d%002d",
            loc->tm_year+1900, loc->tm_mon+1, loc->tm_mday,
            loc->tm_hour, loc->tm_min, loc->tm_sec);
    
    // 3. 소유자/그룹
    pw_p = getpwuid(f_st.st_uid);
    gr_p = getgrgid(f_st.st_gid);
    sprintf(gFileOwner, "%s", pw_p->pw_name);
    sprintf(gFileGroup, "%s", gr_p->gr_name);
    
    // 4. 권한 (rwxrwxrwx → 777)
    // 비트 연산으로 8진수 권한 계산
    for (i=0; i<9; i++)
        amode[i] = (f_st.st_mode & j) ? xtbl[i]: '0';
    
    // 권한 합산 (4+2+1 방식)
    for (i=0; i<3; i++) {
        k = 0;
        for (j=0; j<3; j++) {
            strncpy(tmpchar, &amode[i*3+j], 1);
            k += atoi(tmpchar);
        }
        sprintf(rmode, "%s%d", rmode, k);
    }
    
    sprintf(gFileMode, "%s", rmode);
}

## 3. 파일 송수신 내부 동작

### 3.1 파일 수신 (ProcFileRcv)

**위치**: ecams_svr.c:889-1012

```c
int ProcFileRcv(Sock, RdBuf, RdLng)
{
    switch (uhead->CF) {
        
        // ===== 1단계: FirstChain - 파일 생성 =====
        case dfFirstChain:
            // 1.1 파일명 추출
            sprintf(filename, "%s", &RdBuf[hdlng+dfRemoteOffset]);
            
            // 1.2 TAR 파일명 저장
            if (gProcWork == 'Z')
                strcpy(tarFilename, filename);
            if (gProcWork == 'X')
                strcpy(idxFilename, filename);
            
            // 1.3 디렉토리 체크 & 생성
            if (access(strDir, F_OK) != 0) {
                sprintf(gszLogMsg, "No Such Directory [%s]\n", filename);
                Agent_log(gszLogMsg);
                
                sprintf(gszLogMsg, "Make Directory!!!! [%s]\n", strDir);
                Agent_log(gszLogMsg);
                
                Local_Dir_Make(strDir);  // 재귀 생성
            }
            
            // 1.4 파일 생성
            gRcvFd = open(filename, O_RDWR|O_CREAT|O_TRUNC, 0666);
            if (gRcvFd < 0) {
                memset(statbuf, 0x00, sizeof(statbuf));
                WorkResultSendData(Sock, statbuf, dfCmd_EROR);
                return (-1);
            }
            
            sprintf(gszLogMsg, "START RECEIVE FILE [%s][%d]\n", 
                    filename, gRcvFd);
            Agent_log(gszLogMsg);
            break;
        
        // ===== 2단계: MiddleChain - 데이터 추가 =====
        case dfMiddleChain:
            lseek(gRcvFd, 0L, SEEK_END);
            write(gRcvFd, &RdBuf[hdlng], pulng);
            break;
        
        // ===== 3단계: LastChain - 완료 처리 =====
        case dfLastChain:
            // 3.1 마지막 데이터 쓰기
            if (pulng > 0) {
                lseek(gRcvFd, 0L, SEEK_END);
                write(gRcvFd, &RdBuf[hdlng], pulng);
            }
            
            // 3.2 파일 닫기
            close(gRcvFd);
            
            // 3.3 TAR 처리
            if (gProcWork == 'Z')
                tarFileMake_ext_Z(tarFilename);  // 압축 해제
            
            if (gProcWork == 'X') {
                tarFileMake_comp_X(idxFilename); // 압축 생성
                indexrcvchk = 1;  // 송신 모드 전환
            }
            
            // 3.4 성공 응답
            memset(tmpbuf, 0x00, sizeof(tmpbuf));
            WorkResultSendData(Sock, tmpbuf, dfCmd_DAOK);
            break;
    }
}
```

int FileDataSend(SockFd)
{
    // 1. 파일 읽기
    lseek(gSndFd, gFileOffset, SEEK_SET);
    lng = read(gSndFd, msgbuf, dfMaxFileSize);
    
    // 2. 오프셋 갱신
    gFileOffset += lng;
    
    // 3. Chain 상태 결정
    if (lng < dfMaxFileSize)
        gProcStep = dfLastChain;  // 마지막
    else
        gProcStep = dfMiddleChain; // 중간
    
    // 4. 헤더 생성

    // 5. 전송
    wrtlng = SendDataToSock(SockFd, tmpbuf, makelng);
    
    // 6. 완료 시 파일 닫기
    if (gProcStep == dfLastChain) {
        close(gSndFd);
    }
}

## 6. 모니터링 스크립트

### 6.1 실시간 모니터링

```bash
#!/bin/bash
# /usr/local/bin/ecams_monitor.sh

LOG_FILE="/var/log/ecams/monitor.log"

while true; do
    TIMESTAMP=$(date '+%Y-%m-%d %H:%M:%S')
    
    # 1. 프로세스 수
    TOTAL=$(ps aux | grep ecams_svr | grep -v grep | wc -l)
    
    # 2. 좀비 수
    ZOMBIES=$(ps aux | awk '$8=="Z"' | grep ecams_svr | wc -l)
    
    # 3. 로그 디렉토리 크기 (MB)
    LOG_SIZE=$(du -sm /var/log/ecams | awk '{print $1}')
    
    # 4. 포트 리스닝 확인
    LISTEN=$(netstat -an | grep :29895 | grep LISTEN | wc -l)
    
    # 경고 조건
    if [ $TOTAL -gt 150 ]; then
        echo "[$TIMESTAMP] CRITICAL: Too many processes: $TOTAL" | tee -a $LOG_FILE
    fi
    
    if [ $ZOMBIES -gt 10 ]; then
        echo "[$TIMESTAMP] WARNING: Zombie processes: $ZOMBIES" | tee -a $LOG_FILE
    fi
    
    if [ $LOG_SIZE -gt 1000 ]; then
        echo "[$TIMESTAMP] WARNING: Log size: ${LOG_SIZE}MB" | tee -a $LOG_FILE
    fi
    
    if [ $LISTEN -eq 0 ]; then
        echo "[$TIMESTAMP] CRITICAL: Agent not listening!" | tee -a $LOG_FILE
    fi
    
    # 정상 상태 기록 (5분마다)
    if [ $(date +%M) = "00" ] || [ $(date +%M) = "30" ]; then
        echo "[$TIMESTAMP] OK: Proc=$TOTAL, Zombie=$ZOMBIES, Log=${LOG_SIZE}MB" >> $LOG_FILE
    fi
    
    sleep 60
done
```
