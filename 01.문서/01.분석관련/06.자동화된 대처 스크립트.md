### 1 빌드 전 환경 점검 스크립트

#!/bin/bash
# check_build_env.sh

echo "=== ECAMS Agent Build Environment Check ==="

# 1. OS 확인
OS=$(uname -s)
echo "OS: $OS"

# 2. 컴파일러 확인
if command -v gcc &> /dev/null; then
    echo "GCC: $(gcc --version | head -n1)"
    CC=gcc
elif command -v cc &> /dev/null; then
    echo "CC: $(cc -V 2>&1 | head -n1)"
    CC=cc
else
    echo "ERROR: No C compiler found!"
    exit 1
fi

# 3. 필수 헤더 확인
HEADERS=(
    "sys/socket.h"
    "netinet/in.h"
    "netdb.h"
    "sys/time.h"
)

for header in "${HEADERS[@]}"; do
    if echo "#include <$header>" | $CC -E - &> /dev/null; then
        echo "✓ $header found"
    else
        echo "✗ $header NOT found"
    fi
done

# 4. 필수 라이브러리 확인
LIBS=("nsl" "socket" "xnet")
for lib in "${LIBS[@]}"; do
    if $CC -l$lib 2>&1 | grep -q "cannot find"; then
        echo "✗ lib$lib NOT found (may need to install)"
    else
        echo "✓ lib$lib found (or not required)"
    fi
done

# 5. autoconf 버전
if command -v autoconf &> /dev/null; then
    echo "Autoconf: $(autoconf --version | head -n1)"
else
    echo "WARNING: autoconf not found"
fi

echo "=== Check Complete ==="
```

### 2 자동 패치 스크립트

```bash
#!/bin/bash
# autopatch.sh - OS별 자동 패치

OS=$(uname -s)

case "$OS" in
    SunOS)
        echo "Applying Solaris patches..."
        
        # Makefile.in 수정
        sed -i 's/LIBS\s*=.*/LIBS = -lxnet -lnsl -lsocket -lresolv/' Makefile.in
        
        # PATH 설정
        export PATH=/usr/xpg4/bin:$PATH
        ;;
        
    AIX)
        echo "Applying AIX patches..."
        
        # configure에 xlc 옵션 추가
        export CC="xlc -q32"
        export CFLAGS="-qlanglvl=extc99 -g"
        ;;
        
    HP-UX)
        echo "Applying HP-UX patches..."
        
        export CC="cc -Ae"
        export CFLAGS="-g +DA2.0W"
        export LIBS="-lxnet"
        ;;
        
    Linux)
        echo "Linux detected - no special patches needed"
        ;;
        
    *)
        echo "Unknown OS: $OS"
        ;;
esac

echo "Patch applied. Now run ./configure"
```

### 3 일괄 빌드 스크립트

```bash
#!/bin/bash
# build.sh - 전체 빌드 자동화

set -e  # 에러 시 중단

echo "=== ECAMS Agent Build Script ==="

# 1. Clean
make clean 2>/dev/null || true
rm -f config.cache config.status

# 2. Configure
if [ ! -f ./configure ]; then
    echo "Running autoconf..."
    autoconf
fi

echo "Running configure..."
./configure

# 3. Build
echo "Compiling..."
make

# 4. Verify
if [ -f ecams_svr ]; then
    echo "✓ Build successful: ecams_svr"
    ls -lh ecams_svr
else
    echo "✗ Build failed: ecams_svr not found"
    exit 1
fi

echo "=== Build Complete ==="
```

### 4 통합 일괄 빌드 스크립트 (새롭게 추가)

위 3개 스크립트(check_build_env.sh, autopatch.sh, build.sh)를 하나로 통합하여 **단일 스크립트 실행으로 전체 빌드 프로세스**를 수행합니다.

#### 통합 스크립트 구조

```
┌─────────────────────────────────────────────────────────┐
│                    build_all.sh                         │
├─────────────────────────────────────────────────────────┤
│  1단계: 환경 점검 (check_build_env.sh 로직)             │
│         - OS/컴파일러 확인                              │
│         - 필수 헤더/라이브러리 검증                     │
├─────────────────────────────────────────────────────────┤
│  2단계: OS별 패치 (autopatch.sh 로직)                   │
│         - Solaris: -lxnet -lnsl -lsocket 추가          │
│         - AIX: xlc 옵션 설정                            │
│         - HP-UX: cc -Ae 옵션 설정                       │
├─────────────────────────────────────────────────────────┤
│  3단계: 빌드 수행 (build.sh 로직)                       │
│         - clean → configure → make → 검증              │
└─────────────────────────────────────────────────────────┘
```

#### 통합 스크립트

```bash
#!/bin/bash
# build_all.sh - 환경점검 + 패치 + 빌드 통합 스크립트
#
# 사용법: ./build_all.sh [옵션]
#   옵션:
#     --skip-check    환경 점검 건너뛰기
#     --skip-patch    OS 패치 건너뛰기
#     --clean-only    clean만 수행
#     --help          도움말 출력

set -e  # 에러 시 즉시 중단

#===================================================================
# 변수 초기화
#===================================================================
SKIP_CHECK=false
SKIP_PATCH=false
CLEAN_ONLY=false
BUILD_LOG="build_$(date +%Y%m%d_%H%M%S).log"

#===================================================================
# 옵션 파싱
#===================================================================
while [[ $# -gt 0 ]]; do
    case "$1" in
        --skip-check) SKIP_CHECK=true; shift ;;
        --skip-patch) SKIP_PATCH=true; shift ;;
        --clean-only) CLEAN_ONLY=true; shift ;;
        --help)
            echo "Usage: $0 [--skip-check] [--skip-patch] [--clean-only]"
            exit 0
            ;;
        *) echo "Unknown option: $1"; exit 1 ;;
    esac
done

#===================================================================
# 로깅 함수
#===================================================================
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a "$BUILD_LOG"
}

log_error() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] ERROR: $1" | tee -a "$BUILD_LOG"
}

log_success() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] ✓ $1" | tee -a "$BUILD_LOG"
}

log_fail() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] ✗ $1" | tee -a "$BUILD_LOG"
}

#===================================================================
# 1단계: 환경 점검 (check_build_env.sh 통합)
#===================================================================
check_build_env() {
    log "=========================================="
    log "1단계: 빌드 환경 점검"
    log "=========================================="

    # OS 확인
    OS=$(uname -s)
    log "OS: $OS ($(uname -r))"

    # 컴파일러 확인
    if command -v gcc &> /dev/null; then
        log_success "GCC: $(gcc --version | head -n1)"
        export CC=gcc
    elif command -v cc &> /dev/null; then
        log_success "CC: $(cc -V 2>&1 | head -n1 || echo 'cc found')"
        export CC=cc
    else
        log_fail "C 컴파일러를 찾을 수 없습니다!"
        exit 1
    fi

    # 필수 헤더 확인
    HEADERS=("sys/socket.h" "netinet/in.h" "netdb.h" "sys/time.h" "arpa/inet.h")
    MISSING_HEADERS=()

    for header in "${HEADERS[@]}"; do
        if echo "#include <$header>" | $CC -E - &> /dev/null; then
            log_success "$header"
        else
            log_fail "$header NOT found"
            MISSING_HEADERS+=("$header")
        fi
    done

    if [ ${#MISSING_HEADERS[@]} -gt 0 ]; then
        log_error "누락된 헤더: ${MISSING_HEADERS[*]}"
        log_error "개발 패키지 설치 필요 (예: build-essential, glibc-devel)"
        exit 1
    fi

    # autoconf 확인
    if command -v autoconf &> /dev/null; then
        log_success "Autoconf: $(autoconf --version | head -n1)"
    else
        log "WARNING: autoconf 미설치 (configure 파일이 있으면 무관)"
    fi

    log_success "환경 점검 완료"
}

#===================================================================
# 2단계: OS별 자동 패치 (autopatch.sh 통합)
#===================================================================
apply_os_patch() {
    log "=========================================="
    log "2단계: OS별 자동 패치"
    log "=========================================="

    OS=$(uname -s)

    case "$OS" in
        SunOS)
            log "Solaris 패치 적용 중..."

            # Makefile.in 수정 (네트워크 라이브러리)
            if [ -f Makefile.in ]; then
                if ! grep -q "\-lxnet" Makefile.in; then
                    sed -i 's/LIBS\s*=.*/LIBS = -lxnet -lnsl -lsocket -lresolv/' Makefile.in
                    log_success "Makefile.in에 Solaris 라이브러리 추가"
                fi
            fi

            # XPG4 PATH 우선
            export PATH=/usr/xpg4/bin:$PATH
            log_success "PATH에 /usr/xpg4/bin 추가"
            ;;

        AIX)
            log "AIX 패치 적용 중..."

            export CC="xlc -q32"
            export CFLAGS="-qlanglvl=extc99 -g"
            log_success "AIX xlc 32bit 옵션 설정"
            ;;

        HP-UX)
            log "HP-UX 패치 적용 중..."

            export CC="cc -Ae"
            export CFLAGS="-g +DA2.0W"
            export LIBS="-lxnet"
            log_success "HP-UX cc 옵션 설정"
            ;;

        Linux)
            log_success "Linux - 특별한 패치 불필요"
            ;;

        *)
            log "알 수 없는 OS: $OS (기본 설정 사용)"
            ;;
    esac

    log_success "OS 패치 완료"
}

#===================================================================
# 3단계: 빌드 수행 (build.sh 통합)
#===================================================================
do_build() {
    log "=========================================="
    log "3단계: 빌드 수행"
    log "=========================================="

    # 3-1. Clean
    log "이전 빌드 정리 중..."
    make clean 2>/dev/null || true
    rm -f config.cache config.status config.log
    log_success "Clean 완료"

    # Clean만 수행하는 경우 종료
    if [ "$CLEAN_ONLY" = true ]; then
        log_success "Clean only 모드 - 빌드 생략"
        return 0
    fi

    # 3-2. Configure
    if [ ! -f ./configure ]; then
        if command -v autoconf &> /dev/null; then
            log "configure 파일 생성 중..."
            autoconf
            log_success "autoconf 완료"
        else
            log_fail "configure 파일이 없고 autoconf도 설치되지 않음"
            exit 1
        fi
    fi

    log "configure 실행 중..."
    ./configure 2>&1 | tee -a "$BUILD_LOG"

    if [ ${PIPESTATUS[0]} -ne 0 ]; then
        log_fail "configure 실패 - config.log 확인 필요"
        exit 1
    fi
    log_success "configure 완료"

    # 3-3. Make
    log "컴파일 중..."
    make 2>&1 | tee -a "$BUILD_LOG"

    if [ ${PIPESTATUS[0]} -ne 0 ]; then
        log_fail "컴파일 실패"
        exit 1
    fi
    log_success "컴파일 완료"

    # 3-4. 빌드 결과 확인
    # Makefile에서 mv로 설치 경로로 이동할 수 있으므로 여러 경로 확인
    INSTALL_PATHS=(
        "./ecams_svr"
        "/SW2/polaris/bin/ecams_svr"
        "../../../bin/ecams_svr"
        "$(dirname $(pwd))/bin/ecams_svr"
    )

    FOUND_PATH=""
    for path in "${INSTALL_PATHS[@]}"; do
        if [ -f "$path" ]; then
            FOUND_PATH="$path"
            break
        fi
    done

    if [ -n "$FOUND_PATH" ]; then
        log_success "빌드 성공!"
        log "  파일: $FOUND_PATH"
        log "  크기: $(ls -lh "$FOUND_PATH" | awk '{print $5}')"
        log "  MD5:  $(md5sum "$FOUND_PATH" 2>/dev/null || md5 "$FOUND_PATH" 2>/dev/null || echo 'N/A')"
    else
        log_fail "ecams_svr 파일을 찾을 수 없음"
        log "  확인 경로: ${INSTALL_PATHS[*]}"
        exit 1
    fi
}

#===================================================================
# 메인 실행
#===================================================================
main() {
    log "=========================================="
    log "ECAMS Agent 통합 빌드 시작"
    log "=========================================="
    log "로그 파일: $BUILD_LOG"

    # 1단계: 환경 점검
    if [ "$SKIP_CHECK" = false ]; then
        check_build_env
    else
        log "환경 점검 건너뜀 (--skip-check)"
    fi

    # 2단계: OS 패치
    if [ "$SKIP_PATCH" = false ]; then
        apply_os_patch
    else
        log "OS 패치 건너뜀 (--skip-patch)"
    fi

    # 3단계: 빌드
    do_build

    log "=========================================="
    log "eCAMS Agent Build Success"
    log "=========================================="
}

# 스크립트 실행
main "$@"
```

## 8. 트러블슈팅 체크리스트

- [ ] 1. `./configure` 실행했는가?
- [ ] 2. `config.log` 에서 실패 원인 확인
- [ ] 3. 필수 헤더 파일 모두 설치되었는가?
